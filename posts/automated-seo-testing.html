import pytest
from playwright.sync_api import Page

class TestSEOBasics:
    def test_page_has_valid_title(self, page: Page):
        """测试：页面有有效�?title 标签"""
        page.goto("/")
        title = page.title()
        
        assert len(title) > 0, "页面没有 title 标签"
        assert len(title) <= 60, f"title 过长: {len(title)} 字符"
        assert len(title) >= 30, f"title 过短: {len(title)} 字符"
    
    def test_page_has_meta_description(self, page: Page):
        """测试：页面有 meta description"""
        page.goto("/")
        desc = page.locator('meta[name="description"]').get_attribute("content")
        
        assert desc, "页面没有 meta description"
        assert 50 <= len(desc) <= 160, f"description 长度不合�? {len(desc)}"
    
    def test_images_have_alt_text(self, page: Page):
        """测试：所有图片都�?alt 文本"""
        page.goto("/")
        images = page.locator("img").all()
        
        images_without_alt = []
        for img in images:
            alt = img.get_attribute("alt")
            if not alt or alt.strip() == "":
                src = img.get_attribute("src")
                images_without_alt.append(src)
        
        assert len(images_without_alt) == 0, \
            f"以下图片缺少 alt 文本: {images_without_alt}"
    
    def test_no_broken_links(self, page: Page):
        """测试：页面没有死�?""
        page.goto("/")
        links = page.locator("a[href]").all()
        
        for link in links:
            href = link.get_attribute("href")
            if href.startswith(("http", "https")):
                # 发起请求验证链接
                response = page.request.get(href)
                assert response.ok, f"链接失效: {href} (状态码: {response.status})"

from lighthouse import lighthouse

def test_lighthouse_scores(page: Page):
    """测试：Lighthouse 各项评分达标"""
    page.goto("/")
    
    runner = lighthouse.LighthouseRunner(
        url=page.url,
        form_factor="mobile",
        output="json"
    )
    result = runner.run()
    
    # 验证各项评分
    assert result["categories"]["performance"]["score"] >= 0.8, \
        f"性能评分过低: {result['categories']['performance']['score']}"
    assert result["categories"]["accessibility"]["score"] >= 0.9, \
        f"可访问性评分过�? {result['categories']['accessibility']['score']}"
    assert result["categories"]["seo"]["score"] >= 0.9, \
        f"SEO 评分过低: {result['categories']['seo']['score']}"
